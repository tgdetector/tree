import turtle import time import math import colorsys def setup_screen(): screen = turtle.Screen() screen.setup(width=900, height=800) screen.bgcolor("#0d1117") # Elegant dark background screen.title("Growing Colorful Tree") return screen def show_intro(pen): pen.penup() pen.goto(0, 0) pen.color("#f0f6fc") # Typing / fade-in effect for "Let's Start..." message = "Let's Start..." for i in range(1, len(message) + 1): pen.clear() pen.write(message[:i], align="center", font=("Poppins", 36, "bold")) time.sleep(0.12) time.sleep(1) pen.clear() def draw_branch(t, branch_len, thickness, hue): """ Recursively draws branches that slowly grow and shift in color. """ if branch_len < 10: # Draw a small blooming leaf/flower at the tip t.color("#ff79c6") t.dot(branch_len * 0.8 + 4) return # Convert HSV color to RGB for smooth rainbow gradients r, g, b = colorsys.hsv_to_rgb(hue % 1.0, 0.85, 0.95) t.pencolor(r, g, b) t.pensize(thickness) # Grow forward slowly t.forward(branch_len) # Right subtree angle_right = 24 angle_left = 22 reduction = 0.78 t.right(angle_right) draw_branch(t, branch_len * reduction, thickness * 0.72, hue + 0.07) # Left subtree t.left(angle_right + angle_left) draw_branch(t, branch_len * reduction, thickness * 0.72, hue + 0.12) # Return to the branch base t.right(angle_left) t.penup() t.backward(branch_len) t.pendown() def main(): screen = setup_screen() # Create turtle for the intro text text_pen = turtle.Turtle() text_pen.hideturtle() text_pen.speed(0) # Play intro show_intro(text_pen) # Create turtle for growing the tree tree_turtle = turtle.Turtle() tree_turtle.hideturtle() tree_turtle.speed(3) # Speed 1-3 gives a natural "growing" animation tree_turtle.left(90) # Point upwards tree_turtle.penup() tree_turtle.goto(0, -320) # Start at bottom center tree_turtle.pendown() # Small title watermark at top title_pen = turtle.Turtle() title_pen.hideturtle() title_pen.penup() title_pen.goto(0, 340) title_pen.color("#8b949e") title_pen.write("Nature's Fractal", align="center", font=("Arial", 16, "italic")) # Grow tree (initial trunk length: 125, initial thickness: 11, starting hue: 0.1) draw_branch(tree_turtle, branch_len=125, thickness=11, hue=0.08) # Finished text title_pen.goto(0, -370) title_pen.color("#58a6ff") title_pen.write("Growth Complete ✨", align="center", font=("Arial", 14, "bold")) screen.mainloop() if __name__ == "__main__": main()
